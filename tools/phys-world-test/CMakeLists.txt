# Host build of tools/phys-world-test.cpp against entry/src/main/cpp/JoltPhysics.
cmake_minimum_required(VERSION 3.16)
project(phys_world_test CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
get_filename_component(CPP_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../../entry/src/main/cpp ABSOLUTE)
set(PHYSICS_REPO_ROOT ${CPP_DIR}/JoltPhysics)
set(OBJECT_LAYER_BITS 16)
include(${PHYSICS_REPO_ROOT}/Jolt/Jolt.cmake)
target_compile_options(Jolt PRIVATE -fno-exceptions -fno-rtti)
target_include_directories(Jolt PUBLIC ${PHYSICS_REPO_ROOT})
add_executable(phys_world_test ../phys-world-test.cpp)
target_include_directories(phys_world_test PRIVATE ${CPP_DIR})
target_compile_options(phys_world_test PRIVATE -fno-exceptions -fno-rtti -Wall -Wextra)
target_link_libraries(phys_world_test PRIVATE Jolt)
